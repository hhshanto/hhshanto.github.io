{%- comment -%} Related posts for the current document. The previous version looped over `site.posts`, but this site keeps all writing in collections and has no _posts directory, so it could never match anything. This version searches the current document's own collection and ranks candidates in two passes: 1. documents sharing at least one tag 2. documents from the same subcategory folder Nothing is emitted at all when there are no matches, so a post without siblings does not grow an empty "Related Posts" heading. {%- endcomment -%} {%- assign max_related = 3 -%} {%- assign pool = site[page.collection] | sort: 'date' | reverse -%} {%- assign segments = page.path | split: '/' -%} {%- assign current_group = '' -%} {%- if segments.size > 2 -%} {%- assign current_group = segments[1] -%} {%- endif -%} {%- comment -%} Membership is tracked as a delimited string because Liquid has no way to append to an array. Keys are wrapped in '|' on both sides so one URL can never match as a prefix of another. {%- endcomment -%} {%- assign picked = '|' -%} {%- assign picked_count = 0 -%} {%- comment -%} Pass 1: shared tags. {%- endcomment -%} {%- if page.tags and page.tags.size > 0 -%} {%- for doc in pool -%} {%- if picked_count < max_related and doc.url != page.url -%} {%- assign key = doc.url | prepend: '|' | append: '|' -%} {%- unless picked contains key -%} {%- assign shares_tag = false -%} {%- for tag in doc.tags -%} {%- if page.tags contains tag -%} {%- assign shares_tag = true -%} {%- endif -%} {%- endfor -%} {%- if shares_tag -%} {%- assign picked = picked | append: doc.url | append: '|' -%} {%- assign picked_count = picked_count | plus: 1 -%} {%- endif -%} {%- endunless -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- comment -%} Pass 2: same subcategory folder. {%- endcomment -%} {%- if current_group != '' -%} {%- for doc in pool -%} {%- if picked_count < max_related and doc.url != page.url -%} {%- assign key = doc.url | prepend: '|' | append: '|' -%} {%- unless picked contains key -%} {%- assign doc_segments = doc.path | split: '/' -%} {%- if doc_segments.size > 2 and doc_segments[1] == current_group -%} {%- assign picked = picked | append: doc.url | append: '|' -%} {%- assign picked_count = picked_count | plus: 1 -%} {%- endif -%} {%- endunless -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- if picked_count > 0 -%}
